## 🔍 实时监控

### 监控脚本

```bash
#!/bin/bash
# 文件名：monitor_progress.sh
# 用途：实时监控配置进度

echo "=== openvela 配置进度监控 ==="
echo ""

# 1. 系统环境
echo "1. 系统环境："
wsl -d Ubuntu-24.04 -- bash -c "whoami && uname -r" 2>/dev/null || echo "  ❌ WSL2 未运行"
echo ""

# 2. 开发工具
echo "2. 开发工具："
wsl -d Ubuntu-24.04 -- bash -c '
tools=("gcc" "arm-none-eabi-gcc" "git" "python3" "make" "cmake" "minicom" "repo")
for tool in "${tools[@]}"; do
    if command -v "$tool" &> /dev/null; then
        echo "  ✅ $tool"
    else
        echo "  ❌ $tool"
    fi
done
'
echo ""

# 3. 工程状态
echo "3. 工程状态："
wsl -d Ubuntu-24.04 -- bash -c "
if [ -d ~/contest2026_325_junweiyanjiuyuan ]; then
    cd ~/contest2026_325_junweiyanjiuyuan
    dir_count=\$(ls -d */ 2>/dev/null | wc -l)
    missing=\$(repo status 2>/dev/null | grep -c 'missing')
    echo "  📂 目录数量: \$dir_count"
    echo "  ⏳ 待同步项目: \$missing"
else
    echo "  ❌ 工程目录不存在"
fi
"
echo ""

# 4. 配置完成度
echo "4. 配置完成度："
wsl -d Ubuntu-24.04 -- bash -c '
total=0
passed=0

# 检查工具
tools=("gcc" "arm-none-eabi-gcc" "git" "python3" "make" "cmake" "minicom" "repo")
for tool in "${tools[@]}"; do
    total=$((total + 1))
    if command -v "$tool" &> /dev/null; then
        passed=$((passed + 1))
    fi
done

# 检查工程
if [ -d ~/contest2026_325_junweiyanjiuyuan ]; then
    total=$((total + 1))
    cd ~/contest2026_325_junweiyanjiuyuan
    if [ $(repo status 2>/dev/null | grep -c "missing") -eq 0 ]; then
        passed=$((passed + 1))
    fi
fi

percent=$((passed * 100 / total))
echo "  进度: \$percent% (\$passed/\$total)"
'

echo ""
echo "=== 监控完成 ==="
```

### 使用监控脚本

```bash
# 保存脚本
vim monitor_progress.sh
chmod +x monitor_progress.sh

# 运行监控
./monitor_progress.sh

# 定时监控（每30秒）
watch -n 30 ./monitor_progress.sh
```
